    if(dp[time]>0){
        return dp[time];
    }

#define MAXPASTURE 150

int n;

double diam[MAXPASTURE];

Point pt[MAXPASTURE];

double
ptdist(Point a, Point b)
{
    return sqrt((double)(b.x-a.x)*(b.x-a.x)+(double)(b.y-a.y)*(b.y-a.y));
}

/* mark the field containing pasture i with number m */
void
mark(int i, int m)
{
    int j;
    if(field[i] != -1) {
        assert(field[i] == m);
        return;
    }

    field[i] = m;
    for(j=0; j<n; j++)
        if(dist[i][j] < INF/2)
            mark(j, m);
}

main(void)
{
    FILE *fin, *fout;

    double newdiam, d;

    fin = fopen("cowtour.in", "r");
    fout = fopen("cowtour.out", "w");
    assert(fin != NULL && fout != NULL);

    for(i=0; i<n; i++)
        fscanf(fin, "%d %d\n", &pt[i].x, &pt[i].y);
        
    for(i=0; i<n; i++) {
        for(j=0; j<n; j++) {
            c = getc(fin);
            if(i == j)
                dist[i][j] = 0;
            else if(c == '0')
                dist[i][j] = INF;        /* a lot */
            else
                dist[i][j] = ptdist(pt[i], pt[j]);
        }
        assert(getc(fin) == '\n');
    }

    /* Floyd-Warshall all pairs shortest paths */
    for(k=0; k<n; k++)
    for(i=0; i<n; i++)
    for(j=0; j<n; j++)
        if(dist[i][k]+dist[k][j] < dist[i][j])
            dist[i][j] = dist[i][k]+dist[k][j];

    /* mark fields */
    for(i=0; i<n; i++)
        field[i] = -1;
    for(i=0; i<n; i++)
        if(field[i] == -1)
            mark(i, nfield++);

    /* find worst diameters involving pasture i, and for whole field */
    for(i=0; i<n; i++) {
        for(j=0; j<n; j++)
            if(diam[i] < dist[i][j] && dist[i][j] < INF/2)
                diam[i] = dist[i][j];
        if(diam[i] > fielddiam[field[i]])
            fielddiam[field[i]] = diam[i];
    }

    /* consider a new path between i and j */
    newdiam = INF;
    for(i=0; i<n; i++)
    for(j=0; j<n; j++) {
        if(field[i] == field[j])
            continue;

        d = diam[i]+diam[j]+ptdist(pt[i], pt[j]);
        if(d < fielddiam[field[i]])
            d = fielddiam[field[i]];
        if(d < fielddiam[field[j]])
            d = fielddiam[field[j]];
    
        if(d < newdiam)
            newdiam = d;
    }

    fprintf(fout, "%.6lf\n", newdiam);
    exit(0);
}

const double INF = 1e18;

double compute_dist(const vector<int>& a, const vector<int>& b) {
    double dx = a[0] - b[0];
    double dy = a[1] - b[1];
    return sqrt(dx * dx + dy * dy);
}

void floyd_warshall(vector<vector<double>>& dist) {
    int n = dist.size();

    for (int k = 0; k < n; k++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (dist[i][k] < INF && dist[k][j] < INF) {
                    dist[i][j] =
                        min(dist[i][j], dist[i][k] + dist[k][j]);
                }
            }
        }
    }
}

                    compDiameter[comp[j]],

int N;
vector<pair<int,int>> holes;
int partner[12]; // partner[i] = j means wormhole i is paired with j

// Returns the next wormhole Bessie hits after exiting wormhole idx
// (she exits at holes[partner[idx]], walks right)
int walk_right(int idx) {
    int px = holes[partner[idx]].first;
    int py = holes[partner[idx]].second;
    int best = -1, bestx = INT_MAX;
    for (int i = 0; i < N; i++) {
        if (holes[i].second == py && holes[i].first > px) {
            if (holes[i].first < bestx) {
                bestx = holes[i].first;
                best = i;
            }
        }
    }
    return best; // -1 if she walks off the map
}

bool has_cycle() {
    // Try starting Bessie just to the left of each wormhole
    // (simulates her walking into that wormhole first)
    for (int start = 0; start < N; start++) {
        // Simulate entering wormhole 'start'
        int visited[12] = {};
        int cur = start;
        while (cur != -1) {
            if (visited[cur]) return true; // cycle!
            visited[cur] = 1;
            cur = walk_right(cur); // exit partner[cur], walk to next hole
        }
    }
    return false;
}

    // Find next unpaired wormhole
    while (first_unused < N && partner[first_unused] != -1)

    }
}

long long findMedian(long long n) {
    long long k = (n * n + 1) / 2; // median position
    long long lo = 1, hi = n * n;
    while (lo < hi) {
        long long mid = lo + (hi - lo) / 2;
        if (countLessEqual(mid, n) < k)
            lo = mid + 1;
        else
            hi = mid;
    }
    return lo;
}

vector<vector<ll>> buildPrefixSum(const vector<vector<ll>> &arr) {
    ll N = arr.size();
    ll M = arr[0].size();
    vector<vector<ll>> pref(N+1, vector<ll>(M+1, 0));

    for (ll i = 1; i <= N; i++) {
        for (ll j = 1; j <= M; j++) {
            pref[i][j] = arr[i-1][j-1] 
                       + pref[i-1][j] 
                       + pref[i][j-1] 
                       - pref[i-1][j-1];
        }
    }
    return pref;
}

Solution: 
The solution is to make bipartite graph. Then do a dfs (or even bfs will work) to go through all the nodes possible when starting from a particular node, and mark them as visited, then only visit the unvisited nodes, find the connected components and the answer will be total connected components -1.
